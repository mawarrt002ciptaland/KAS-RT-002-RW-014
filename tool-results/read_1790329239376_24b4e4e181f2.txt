   119→/** Group transaksi by month for a given year (ascending Jan→Des, only months with items). */
   120→function groupByMonth(items: Transaksi[], year: number): MonthGroup[] {
   121→  const groups: MonthGroup[] = [];
   122→  for (let i = 0; i < 12; i++) {
   123→    groups.push({ monthIdx: i, monthName: BULAN_ID[i], items: [], total: 0 });
   124→  }
   125→  for (const t of items) {
   126→    const d = new Date(t.tanggal);
   127→    if (Number.isNaN(d.getTime())) continue;
   128→    if (d.getFullYear() !== year) continue;
   129→    groups[d.getMonth()].items.push(t);
   130→    groups[d.getMonth()].total += t.nominal || 0;
   131→  }
   132→  // Sort items within each month ascending by date for stable display
   133→  for (const g of groups) {
   134→    g.items.sort((a, b) => +new Date(a.tanggal) - +new Date(b.tanggal));
   135→  }
   136→  return groups.filter((g) => g.items.length > 0);
   137→}

   164→

   196→  }, [items, mounted, year]);

   198→  const monthGroups = useMemo(() => groupByMonth(items, year), [items, year]);
   199→  const yearTotal = useMemo(
   200→    () => monthGroups.reduce((s, g) => s + g.total, 0),
   201→    [monthGroups],
   202→  );
   203→  const yearCount = useMemo(
   204→    () => monthGroups.reduce((s, g) => s + g.items.length, 0),
   205→    [monthGroups],
   206→  );

   208→  const bulanIni = useMemo(() => {
   209→    const now = new Date();

   211→    const m = now.getMonth();

   214→        const d = new Date(t.tanggal);

   216→      })

   224→      if (next.has(idx)) next.delete(idx);
   225→      else next.add(idx);
   226→      return next;

   237→    setDeleteTarget(null);
   238→    setDetailItem(null);

   261→      />
   262→
   263→      {/* Summary StatCards */}
   264→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-4">
   265→        <StatCard title={`Total ${title}`} tone={tone} value={<RupiahText value={total} />} icon={<Wallet className="h-5 w-5" />} hint={`Akumulasi ${count} transaksi`} />
   266→        <StatCard title="Jumlah Transaksi" tone="neutral" value={count} icon={<Hash className="h-5 w-5" />} hint={kategori !== "all" ? `Filter: ${kategori}` : "Tanpa filter"} />
   267→        <StatCard title="Bulan Ini" tone={tone} value={<RupiahText value={bulanIni} />} icon={<Calendar className="h-5 w-5" />} hint={mounted ? formatTanggalID(new Date()) : "\u00A0"} />

   271→      {/* Filter bar: search + kategori + year selector (mobile stacked) */}
   272→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   273→        <div className="relative flex-1">
   274→          <Search className="pointer-events-none absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   275→          <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Cari keterangan, kode, nama…" className="touch-target pl-9" aria-label="Cari transaksi" />
   276→          {q && (
   277→            <button type="button" onClick={() => setQ("")} className="touch-target absolute top-1/2 right-2 -translate-y-1/2 rounded-md p-1.5 text-muted-foreground hover:bg-accent" aria-label="Hapus pencarian">

   311→        </div>
   312→      </div>

   314→      {/* Monthly grouping */}

   334→          <EmptyState
   335→            icon={<Calendar className="h-6 w-6" />}
   336→            title={`Tidak ada transaksi tahun ${year}`}

   339→          />
   340→        ) : (
   341→          <div className="space-y-4">
   342→            {monthGroups.map((g) => {
   343→              const isCollapsed = collapsed.has(g.monthIdx);
   344→              const rowProps: RowProps = { ...baseRow, items: g.items };
   345→              return (
   346→                <div key={g.monthIdx} className="rounded-xl border bg-card">
   347→                  {/* Sticky month header bar */}
   348→                  <div className="sticky top-16 z-20 flex items-center gap-2 rounded-t-xl border-b bg-card/95 p-3 backdrop-blur supports-[backdrop-filter]:bg-card/80">
   349→                    <button
   350→                      type="button"
   351→                      onClick={() => toggleCollapse(g.monthIdx)}
   352→                      className="touch-target -ml-1 rounded-md p-1.5 hover:bg-accent"
   353→                      aria-label={isCollapsed ? `Buka ${g.monthName}` : `Tutup ${g.monthName}`}
   354→                      aria-expanded={!isCollapsed}

   356→                      {isCollapsed ? <ChevronRight className="h-4 w-4" /> : <ChevronDown className="h-4 w-4" />}
   357→                    </button>
   358→                    <Calendar className="hidden h-4 w-4 shrink-0 text-muted-foreground sm:block" />
   359→                    <h3 className="min-w-0 flex-1 truncate text-sm font-semibold">{g.monthName} {year}</h3>
   360→                    <Badge variant="secondary" className="shrink-0 text-[11px]">{g.items.length} trx</Badge>
   361→                    <p className={`shrink-0 text-sm font-bold tabular-nums ${nominalColor}`}>{sign}{formatRupiah(g.total)}</p>
   362→                  </div>
   363→
   364→                  {/* Body */}
   365→                  {!isCollapsed && (
   366→                    <>
   367→                      {isMobile ? (
   368→                        <MobileCardList {...rowProps} />
   369→                      ) : (
   370→                        <div className="overflow-x-auto scrollbar-thin">
   371→                          <DesktopTable {...rowProps} />
   372→                        </div>
   373→                      )}
   374→                      {/* Monthly subtotal footer */}
   375→                      <div className="flex items-center justify-between rounded-b-xl border-t bg-muted/40 px-3 py-2 text-sm">
   376→                        <span className="font-medium">Total {g.monthName}</span>
   377→                        <span className={`font-bold tabular-nums ${nominalColor}`}>{sign}{formatRupiah(g.total)}</span>
   378→                      </div>
   379→                    </>
   380→                  )}
   381→                </div>
   382→              );
   383→            })}
   384→
   385→            {/* Year total summary */}
   386→            <div className={`flex flex-col gap-2 rounded-xl border-2 p-4 sm:flex-row sm:items-center sm:justify-between ${isPemasukan ? "border-success/40 bg-success/5" : "border-destructive/40 bg-destructive/5"}`}>
   387→              <div className="flex items-center gap-3">
   388→                <span className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-full ${isPemasukan ? "bg-success/15 text-success" : "bg-destructive/10 text-destructive"}`}>
   389→                  <Icon className="h-5 w-5" />
   390→                </span>
   391→                <div>
   392→                  <p className="text-xs text-muted-foreground">Total Tahun {year}</p>
   393→                  <p className="text-sm font-medium">{yearCount} transaksi • {monthGroups.length} bulan aktif</p>
   394→                </div>
   395→              </div>
   396→              <p className={`text-2xl font-bold tabular-nums ${nominalColor}`}>{sign}{formatRupiah(yearTotal)}</p>
   397→            </div>
   398→          </div>
   399→        )}
   400→      </section>
   401→
   402→      {/* Create Dialog */}
   403→      <CreateTransaksiDialog
   404→        open={createOpen}
   405→        onOpenChange={setCreateOpen}
   406→        jenis={jenis}
   407→        kategoriList={kategoriList}
   408→        onDone={refetch}
   409→      />
   410→
   411→      {/* Detail Dialog */}
   412→      <Dialog open={!!detailItem} onOpenChange={(o) => !o && setDetailItem(null)}>
   413→        <DialogContent className="max-h-[90vh] overflow-y-auto">
   414→          {detailItem && (
   415→            <DetailBody
   416→              trx={detailItem}
   417→              isPemasukan={isPemasukan}
   418→              nominalColor={nominalColor}
   419→              sign={sign}
   420→              onPrint={() => toast.info("Membuka dialog cetak kwitansi…")}
   421→              onDelete={() => setDeleteTarget(detailItem)}
   422→            />
   423→          )}
   424→        </DialogContent>
   425→      </Dialog>
   426→
   427→      {/* Delete confirmation */}
   428→      <AlertDialog open={!!deleteTarget} onOpenChange={(o) => !o && setDeleteTarget(null)}>
   429→        <AlertDialogContent>
   430→          <AlertDialogHeader>
   431→            <AlertDialogTitle>Hapus transaksi ini?</AlertDialogTitle>
   432→            <AlertDialogDescription>
   433→              {deleteTarget?.kode} — {deleteTarget?.keterangan}. Tindakan ini tidak dapat dibatalkan. Kwitansi terkait juga akan dihapus.
   434→            </AlertDialogDescription>
   435→          </AlertDialogHeader>
   436→          <AlertDialogFooter>
   437→            <AlertDialogCancel disabled={deleting} className="touch-target">Batal</AlertDialogCancel>
   438→            <AlertDialogAction onClick={handleDelete} disabled={deleting} className="touch-target bg-destructive text-white hover:bg-destructive/90">
   439→              {deleting ? "Menghapus…" : "Hapus"}
   440→            </AlertDialogAction>
   441→          </AlertDialogFooter>
   442→        </AlertDialogContent>
   443→      </AlertDialog>
   444→    </div>
   445→  );
   446→}

   448→/** ====== MOBILE CARD LIST ====== */

   450→  const Icon = isPemasukan ? TrendingUp : TrendingDown;
   451→  return (
   452→    <div className="divide-y">
   453→      {items.map((t) => (
   454→        <div key={t.id} className="flex items-start gap-3 p-3">
   455→          <span className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-full ${isPemasukan ? "bg-success/15 text-success" : "bg-destructive/10 text-destructive"}`}>
   456→            <Icon className="h-5 w-5" />
   457→          </span>
   458→          <div className="min-w-0 flex-1">
   459→            <p className="truncate text-sm font-semibold">{t.keterangan}</p>
   460→            <div className="mt-0.5 flex flex-wrap items-center gap-1.5 text-xs text-muted-foreground">
   461→              <Badge variant="secondary" className="text-[11px]">{t.kategori}</Badge>
   462→              <span>•</span>
   463→              <span className="truncate font-mono">{t.kode}</span>
   464→            </div>
   465→            <p className="mt-1 text-[11px] text-muted-foreground">
   466→              {formatTanggalID(t.tanggal)} • {(isPemasukan ? t.sumber : t.penerima) || "-"}
   467→            </p>
   468→          </div>
   469→          <div className="flex shrink-0 flex-col items-end gap-1">
   470→            <p className={`text-sm font-bold tabular-nums ${nominalColor}`}>{sign}{formatRupiah(t.nominal)}</p>
   471→            <div className="flex gap-1">
   472→              <button onClick={() => onDetail(t)} className="touch-target rounded-md p-1.5 text-muted-foreground hover:bg-accent hover:text-foreground" aria-label="Lihat detail">
   473→                <Eye className="h-4 w-4" />
   474→              </button>
   475→              <button onClick={() => onDelete(t)} className="touch-target rounded-md p-1.5 text-destructive hover:bg-destructive/10" aria-label="Hapus transaksi">
   476→                <Trash2 className="h-4 w-4" />
   477→              </button>
   478→            </div>
   479→          </div>
   480→        </div>
   481→      ))}
   482→    </div>
   483→  );
   484→}

   493→          <TableHead>Kode</TableHead>
   494→          <TableHead>Keterangan</TableHead>
   495→          <TableHead>Kategori</TableHead>
   496→          <TableHead>{isPemasukan ? "Sumber" : "Penerima"}</TableHead>
   497→          <TableHead className="text-right">Nominal</TableHead>
   498→          <TableHead className="text-right">Aksi</TableHead>
   499→        </TableRow>
   500→      </TableHeader>
   501→      <TableBody>
   502→        {items.map((t) => (
   503→          <TableRow key={t.id}>
   504→            <TableCell className="whitespace-nowrap text-sm">{formatTanggalID(t.tanggal)}</TableCell>
   505→            <TableCell className="max-w-[120px] truncate font-mono text-xs text-muted-foreground">{t.kode}</TableCell>
   506→            <TableCell className="max-w-[260px] truncate text-sm font-medium">{t.keterangan}</TableCell>
   507→            <TableCell><Badge variant="secondary" className="text-[11px]">{t.kategori}</Badge></TableCell>
   508→            <TableCell className="text-sm text-muted-foreground">{(isPemasukan ? t.sumber : t.penerima) || "-"}</TableCell>
   509→            <TableCell className={`whitespace-nowrap text-right text-sm font-bold tabular-nums ${nominalColor}`}>{sign}{formatRupiah(t.nominal)}</TableCell>
   510→            <TableCell className="text-right">
   511→              <div className="flex justify-end gap-1">
   512→                <Button variant="ghost" size="icon" onClick={() => onDetail(t)} aria-label="Lihat detail" className="touch-target"><Eye className="h-4 w-4" /></Button>
   513→                <Button variant="ghost" size="icon" onClick={() => onDelete(t)} aria-label="Hapus transaksi" className="touch-target text-destructive hover:bg-destructive/10 hover:text-destructive"><Trash2 className="h-4 w-4" /></Button>
   514→              </div>
   515→            </TableCell>
   516→          </TableRow>
   517→        ))}
   518→      </TableBody>
   519→    </Table>
   520→  );
   521→}

   524→function DetailBody({

   584→/** ====== CREATE DIALOG ====== */
   585→function CreateTransaksiDialog({
   586→  open, onOpenChange, jenis, kategoriList, onDone,
   587→}: {
   588→  open: boolean;
   589→  onOpenChange: (o: boolean) => void;
   590→  jenis: Jenis;
   591→  kategoriList: string[];
   592→  onDone: () => void;
   593→}) {
   594→  const isPemasukan = jenis === "pemasukan";
   595→  const title = isPemasukan ? "Catat Pemasukan" : "Catat Pengeluaran";
   596→
   597→  const [tanggal, setTanggal] = useState<string>(() =>
   598→    typeof window !== "undefined" ? toISODate(new Date()) : "2026-01-01",
   599→  );
   600→  const [kategori, setKategori] = useState<string>(kategoriList[0] ?? "");
   601→  const [keterangan, setKeterangan] = useState("");
   602→  const [nominal, setNominal] = useState(0);
   603→  const [penerimaSumber, setPenerimaSumber] = useState("");
   604→  const [metode, setMetode] = useState<string>("Tunai");
   605→  const [buktiName, setBuktiName] = useState<string>("");
   606→  const [saving, setSaving] = useState(false);
   607→  const [touched, setTouched] = useState(false);
   608→
   609→  function resetForm() {
   610→    setTanggal(typeof window !== "undefined" ? toISODate(new Date()) : "2026-01-01");
   611→    setKategori(kategoriList[0] ?? "");
   612→    setKeterangan("");
   613→    setNominal(0);
   614→    setPenerimaSumber("");
   615→    setMetode("Tunai");
   616→    setBuktiName("");
   617→    setTouched(false);
   618→  }
   619→
   620→  const keteranganInvalid = !keterangan.trim();
   621→  const nominalInvalid = !nominal || nominal <= 0;
   622→  const hasError = touched && (keteranganInvalid || nominalInvalid);
   623→
   624→  async function onSubmit(e: React.FormEvent) {
   625→    e.preventDefault();
   626→    setTouched(true);
   627→    if (keteranganInvalid || nominalInvalid) return;
   628→
   629→    setSaving(true);
   630→    const body: Record<string, unknown> = {
   631→      jenis, kategori, keterangan: keterangan.trim(), nominal, tanggal, metode, buktiUrl: buktiName || undefined,
   632→    };
   633→    if (isPemasukan) body.sumber = penerimaSumber.trim() || undefined;
   634→    else body.penerima = penerimaSumber.trim() || undefined;
   635→
   636→    const r = await postJSON("/api/transaksi", body);
   637→    setSaving(false);
   638→    if (!r.ok) return toast.error(r.error || "Gagal menyimpan");
   639→    toast.success(`${isPemasukan ? "Pemasukan" : "Pengeluaran"} berhasil dicatat`);
   640→    resetForm();
   641→    onOpenChange(false);
   642→    onDone();
   643→  }
   644→
   645→  return (
   646→    <Dialog open={open} onOpenChange={(o) => { onOpenChange(o); if (!o) resetForm(); }}>
   647→      <DialogContent className="max-h-[90vh] overflow-y-auto">
   648→        <DialogHeader>
   649→          <DialogTitle>{title}</DialogTitle>
   650→          <DialogDescription>
   651→            Isi data {title.toLowerCase()} kas RT 002. Kode & kwitansi dibuat otomatis.
   652→          </DialogDescription>
   653→        </DialogHeader>
   654→
   655→        <form onSubmit={onSubmit} className="grid gap-4">
   656→          <div className="grid gap-3 sm:grid-cols-2">
   657→            <div className="grid gap-1.5">
   658→              <Label htmlFor="trx-tanggal">Tanggal</Label>
   659→              <Input id="trx-tanggal" type="date" value={tanggal} onChange={(e) => setTanggal(e.target.value)} className="touch-target" required />
   660→            </div>
   661→            <div className="grid gap-1.5">
   662→              <Label htmlFor="trx-kategori">Kategori</Label>
   663→              <Select value={kategori} onValueChange={setKategori}>
   664→                <SelectTrigger id="trx-kategori" className="touch-target w-full"><SelectValue placeholder="Pilih kategori" /></SelectTrigger>
   665→                <SelectContent>
   666→                  {kategoriList.map((k) => <SelectItem key={k} value={k}>{k}</SelectItem>)}
   667→                </SelectContent>
   668→              </Select>
   669→            </div>
   670→          </div>
   671→
   672→          <div className="grid gap-1.5">
   673→            <Label htmlFor="trx-ket">Keterangan</Label>
   674→            <Input
   675→              id="trx-ket"
   676→              value={keterangan}
   677→              onChange={(e) => setKeterangan(e.target.value)}
   678→              placeholder={isPemasukan ? "Contoh: Iuran bulanan warga Blok Mawar" : "Contoh: Pembelian perlengkapan kebersihan"}
   679→              className="touch-target"
   680→              aria-invalid={hasError && keteranganInvalid}
   681→              required
   682→            />
   683→            {hasError && keteranganInvalid && <p className="text-xs text-destructive">Keterangan wajib diisi.</p>}
   684→          </div>
   685→
   686→          <div className="grid gap-1.5">
   687→            <Label htmlFor="trx-nominal">Nominal (Rp)</Label>
   688→            <div className="relative">
   689→              <span className="pointer-events-none absolute top-1/2 left-3 -translate-y-1/2 text-sm text-muted-foreground">Rp</span>
   690→              <Input
   691→                id="trx-nominal"
   692→                inputMode="numeric"
   693→                value={toThousandInput(nominal)}
   694→                onChange={(e) => setNominal(parseRupiahInput(e.target.value))}
   695→                placeholder="0"
   696→                className="touch-target pl-9 tabular-nums"
   697→                aria-invalid={hasError && nominalInvalid}
   698→                required
   699→              />
   700→            </div>
   701→            {hasError && nominalInvalid && <p className="text-xs text-destructive">Nominal harus &gt; 0.</p>}
   702→            <p className="text-xs text-muted-foreground">{formatRupiah(nominal)}</p>
   703→          </div>
   704→
   705→          <div className="grid gap-3 sm:grid-cols-2">
   706→            <div className="grid gap-1.5">
   707→              <Label htmlFor="trx-penerima">{isPemasukan ? "Sumber" : "Penerima"}</Label>
   708→              <Input id="trx-penerima" value={penerimaSumber} onChange={(e) => setPenerimaSumber(e.target.value)} placeholder={isPemasukan ? "Warga / Donatur" : "Nama penerima / vendor"} className="touch-target" />
   709→            </div>
   710→            <div className="grid gap-1.5">
   711→              <Label htmlFor="trx-metode">Metode</Label>
   712→              <Select value={metode} onValueChange={setMetode}>
   713→                <SelectTrigger id="trx-metode" className="touch-target w-full"><SelectValue placeholder="Pilih metode" /></SelectTrigger>
   714→                <SelectContent>
   715→                  {METODE_LIST.map((m) => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   716→                </SelectContent>
   717→              </Select>
   718→            </div>
   719→          </div>
   720→
   721→          <div className="grid gap-1.5">
   722→            <Label htmlFor="trx-bukti">Bukti Transaksi</Label>
   723→            <label htmlFor="trx-bukti" className="touch-target flex cursor-pointer items-center gap-2 rounded-md border border-dashed bg-muted/40 px-3 py-2 text-sm text-muted-foreground hover:bg-muted">
   724→              <Camera className="h-4 w-4" />
   725→              <span className="truncate">{buktiName || "Pilih foto / ambil dari kamera"}</span>
   726→            </label>
   727→            <Input id="trx-bukti" type="file" accept="image/*" capture="environment" className="hidden" onChange={(e) => setBuktiName(e.target.files?.[0]?.name ?? "")} />
   728→            {buktiName && (
   729→              <button type="button" onClick={() => setBuktiName("")} className="touch-target mt-1 inline-flex w-fit items-center gap-1 text-xs text-destructive">
   730→                <X className="h-3 w-3" /> Hapus berkas
   731→              </button>
   732→            )}
   733→          </div>
   734→
   735→          <DialogFooter>
   736→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)} disabled={saving} className="touch-target">Batal</Button>
   737→            <Button type="submit" disabled={saving} className="touch-target">{saving ? "Menyimpan…" : `Simpan ${title}`}</Button>
   738→          </DialogFooter>
   739→        </form>
   740→      </DialogContent>
   741→    </Dialog>
   742→  );
   743→}
