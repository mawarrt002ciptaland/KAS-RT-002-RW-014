    66→function roleBadgeClass(role: string) {
    67→  switch (role) {
    68→    case "admin": return "bg-primary/15 text-primary";
    69→    case "ketua": return "bg-success/15 text-success";

    72→    default: return "bg-muted text-muted-foreground";
    73→  }
    74→}
    75→function roleLabel(role: string) {
    76→  return ROLE_OPTIONS.find((o) => o.value === role)?.label || role;

    78→function initialsOf(name: string) {
    79→  const parts = name.trim().split(/\s+/);
    80→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
    81→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
    82→}

    84→/** POST /api/upload with FormData field "file". Returns the URL or null on error. */
    85→async function uploadImage(file: File): Promise<string | null> {
    86→  const fd = new FormData();
    87→  fd.append("file", file);
    88→  try {
    89→    const res = await fetch("/api/upload", { method: "POST", body: fd });
    90→    const j = await res.json();
    91→    if (!res.ok) { toast.error(j?.error || "Gagal mengunggah"); return null; }
    92→    return j.url as string;
    93→  } catch {
    94→    toast.error("Gagal terhubung ke server");
    95→    return null;
    96→  }
    97→}
    98→
    99→function isImageUrl(u: string) {
   100→  return /^https?:\/\//.test(u) || u.startsWith("/uploads/");
   101→}

   654→function UserFormInner({ mode, user, onSubmitted, onCancel }: {
   655→  mode: "create" | "edit"; user: UserItem | null;
   656→  onSubmitted: () => void; onCancel: () => void;
   657→}) {
   658→  const [email, setEmail] = useState(user?.email || "");
   659→  const [password, setPassword] = useState("");
   660→  const [nama, setNama] = useState(user?.nama || "");
   661→  const [role, setRole] = useState<Role>((user?.role as Role) || "warga");
   662→  const [telepon, setTelepon] = useState(user?.telepon || "");
   663→  const [showPw, setShowPw] = useState(false);
   664→  const [submitting, setSubmitting] = useState(false);
   665→
   666→  async function submit() {
   667→    if (mode === "create") {
   668→      if (!email) return toast.error("Email wajib diisi");
   669→      if (!nama) return toast.error("Nama wajib diisi");
   670→      if (!password) return toast.error("Password wajib diisi");
   671→      if (password.length < 6) return toast.error("Password minimal 6 karakter");
   672→      setSubmitting(true);
   673→      const r = await postJSON("/api/auth/users", { email, password, nama, role, telepon });
   674→      setSubmitting(false);
   675→      if (!r.ok) return toast.error(r.error);
   676→      toast.success(`User ${email} ditambahkan`);
   677→      onSubmitted();
   678→    } else {
   679→      if (!user) return;
   680→      if (!nama) return toast.error("Nama wajib diisi");
   681→      const body: Record<string, string> = { nama, role, telepon };
   682→      if (password) {
   683→        if (password.length < 6) return toast.error("Password minimal 6 karakter");
   684→        body.password = password;
   685→      }
   686→      setSubmitting(true);
   687→      const r = await patchJSON(`/api/auth/users/${user.id}`, body);
   688→      setSubmitting(false);
   689→      if (!r.ok) return toast.error(r.error);
   690→      toast.success(`User ${user.email} diperbarui`);
   691→      onSubmitted();
   692→    }
   693→  }
   694→
   695→  return (
   696→    <div className="grid gap-3">
   697→      <div className="grid gap-1.5">
   698→        <Label htmlFor="u-email">Email *</Label>
   699→        <Input id="u-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} disabled={mode === "edit"} placeholder="email@rt002mawar.id" className="touch-target" />
   700→        {mode === "edit" && <p className="text-[11px] text-muted-foreground">Email tidak dapat diubah</p>}
   701→      </div>
   702→      <div className="grid gap-1.5">
   703→        <Label htmlFor="u-nama">Nama Lengkap *</Label>
   704→        <Input id="u-nama" value={nama} onChange={(e) => setNama(e.target.value)} placeholder="Nama lengkap" className="touch-target" />
   705→      </div>
   706→      <div className="grid gap-1.5">
   707→        <Label htmlFor="u-pw">{mode === "create" ? "Password *" : "Password"}</Label>
   708→        <div className="relative">
   709→          <Input id="u-pw" type={showPw ? "text" : "password"} value={password} onChange={(e) => setPassword(e.target.value)} placeholder={mode === "edit" ? "Kosongkan jika tidak diubah" : "Min. 6 karakter"} className="touch-target pr-10" />
   710→          <Button type="button" variant="ghost" size="sm" className="absolute right-1 top-1/2 h-8 w-8 -translate-y-1/2 p-0" onClick={() => setShowPw((s) => !s)} aria-label={showPw ? "Sembunyikan password" : "Tampilkan password"}>
   711→            {showPw ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   712→          </Button>
   713→        </div>
   714→      </div>
   715→      <div className="grid gap-1.5">
   716→        <Label htmlFor="u-role">Role</Label>
   717→        <Select value={role} onValueChange={(v) => setRole(v as Role)}>
   718→          <SelectTrigger id="u-role" className="touch-target w-full"><SelectValue /></SelectTrigger>
   719→          <SelectContent>
   720→            {ROLE_OPTIONS.map((o) => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}
   721→          </SelectContent>
   722→        </Select>
   723→      </div>
   724→      <div className="grid gap-1.5">
   725→        <Label htmlFor="u-telepon">Telepon</Label>
   726→        <Input id="u-telepon" type="tel" value={telepon} onChange={(e) => setTelepon(e.target.value)} placeholder="0812xxxxxxx" className="touch-target" />
   727→      </div>
   728→      <DialogFooter>
   729→        <Button variant="outline" onClick={onCancel} className="touch-target">Batal</Button>
   730→        <Button onClick={submit} disabled={submitting} className="touch-target">
   731→          <Save className="h-4 w-4" /> {submitting ? "Menyimpan..." : mode === "create" ? "Tambah" : "Simpan"}
   732→        </Button>
   733→      </DialogFooter>
   734→    </div>
   735→  );

   738→function ResetPasswordDialog({ user, onClose, onDone }: {
   739→  user: UserItem | null; onClose: () => void; onDone: () => void;
   740→}) {
   741→  const [pw, setPw] = useState("");
   742→  const [show, setShow] = useState(false);
   743→  const [submitting, setSubmitting] = useState(false);
   744→
   745→  function handleClose(o: boolean) {
   746→    if (!o) { setPw(""); setShow(false); onClose(); }
   747→  }
   748→
   749→  async function submit() {
   750→    if (!user) return;
   751→    if (pw.length < 6) return toast.error("Password minimal 6 karakter");
   752→    setSubmitting(true);
   753→    const r = await patchJSON(`/api/auth/users/${user.id}`, { password: pw });
   754→    setSubmitting(false);
   755→    if (!r.ok) return toast.error(r.error);
   756→    toast.success(`Password ${user.email} diperbarui`);
   757→    setPw("");
   758→    onDone();
   759→  }
   760→
   761→  return (
   762→    <Dialog open={!!user} onOpenChange={handleClose}>
   763→      <DialogContent className="max-h-[90vh] overflow-y-auto">
   764→        <DialogHeader>
   765→          <DialogTitle>Reset Password</DialogTitle>
   766→          <DialogDescription>
   767→            Set password baru untuk <strong>{user?.email}</strong>
   768→          </DialogDescription>
   769→        </DialogHeader>
   770→        <div className="grid gap-3">
   771→          <div className="grid gap-1.5">
   772→            <Label htmlFor="rp-pw">Password Baru</Label>
   773→            <div className="relative">
   774→              <Input id="rp-pw" type={show ? "text" : "password"} value={pw} onChange={(e) => setPw(e.target.value)} placeholder="Min. 6 karakter" className="touch-target pr-10" />
   775→              <Button type="button" variant="ghost" size="sm" className="absolute right-1 top-1/2 h-8 w-8 -translate-y-1/2 p-0" onClick={() => setShow((s) => !s)} aria-label={show ? "Sembunyikan password" : "Tampilkan password"}>

   777→              </Button>
   778→            </div>
   779→          </div>
   780→          <DialogFooter>
   781→            <Button variant="outline" onClick={() => handleClose(false)} className="touch-target">Batal</Button>
   782→            <Button onClick={submit} disabled={submitting} className="touch-target">
   783→              <KeyRound className="h-4 w-4" /> {submitting ? "Menyimpan..." : "Reset Password"}
   784→            </Button>
   785→          </DialogFooter>
   786→        </div>
   787→      </DialogContent>
   788→    </Dialog>
   789→  );
   790→}
